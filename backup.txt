const TABS = ['calc','fx','unit','age','oa'];
function switchTab(t) {
  document.querySelectorAll('.tab-item').forEach((el,i) =>
    el.classList.toggle('active', TABS[i] === t));
  document.querySelectorAll('.screen').forEach(el =>
    el.classList.remove('active'));

function numFmt(n) {
  if (typeof n !== 'number' || isNaN(n)) return String(n);
  const s = parseFloat(n.toPrecision(12)).toString();
  const [int, dec] = s.split('.');
  const fmtInt = int.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
  return dec !== undefined ? fmtInt + '.' + dec : fmtInt;
}

function colorExpr(s) {
  return s.replace(/([÷×−+])/g, '<span class="op">$1</span>');
}

function tryEval(expr) {
  try {
    const e = expr.replace(/,/g,'').replace(/÷/g,'/').replace(/×/g,'*').replace(/−/g,'-');
    const r = Function('"use strict";return(' + e + ')')();
    return (typeof r === 'number' && isFinite(r)) ? r : null;
  } catch { return null; }
}

function ci(k) {
  if (navigator.vibrate) navigator.vibrate(10);
  if (k === 'C') {
    curExpr = ''; curDisp = '0'; lastRes = null; evaled = false; parenD = 0;
  } else if (k === '()') {
    if (evaled) { curExpr = numFmt(lastRes); evaled = false; }
    const last = curExpr.slice(-1);
    if (parenD === 0 || OPS.includes(last) || last === '(') {
      curExpr += '('; parenD++; curDisp = '(';
    } else {
      curExpr += ')'; parenD = Math.max(0, parenD - 1); curDisp = ')';
    }
  } else if (k === '%') {
    const v = parseFloat(curDisp.replace(/,/g,''));
    if (!isNaN(v)) {
      const r = v / 100;
      curDisp = numFmt(r);
      if (evaled) lastRes = r;
      const li = Math.max(...OPS.map(o => curExpr.lastIndexOf(o)));
      curExpr = li >= 0 ? curExpr.slice(0, li+1) + numFmt(r) : numFmt(r);
    }
  } else if (k === '=') {
    const full = curExpr || curDisp;
    const r = tryEval(full);
    if (r !== null) {
      calcHist.push({ expr: curExpr || curDisp, result: numFmt(r) });
      if (calcHist.length > 5) calcHist.shift();
      curExpr = ''; lastRes = r; curDisp = numFmt(r); evaled = true; parenD = 0;
    } else {
      curDisp = '오류';
    }
  } else if (OPS.includes(k)) {
    if (evaled) { curExpr = numFmt(lastRes) + k; curDisp = k; evaled = false; }
    else {
      const last = curExpr.slice(-1);
      if (OPS.includes(last)) curExpr = curExpr.slice(0,-1) + k;
      else curExpr = (curExpr || curDisp) + k;
      curDisp = k;
    }
  } else {
    if (evaled) { curExpr = ''; curDisp = '0'; lastRes = null; evaled = false; }
    if (k === '.' && curDisp.includes('.')) { /* skip */ }
    else if (k === '.' && (curDisp === '' || OPS.includes(curDisp) || curDisp === '(')) {
      curDisp = '0.';
    } else if ((curDisp === '0' || OPS.includes(curDisp) || curDisp === '(') && k !== '.') {
      curDisp = k;
    } else {
      curDisp += k;
    }
    const li = Math.max(...OPS.map(o => curExpr.lastIndexOf(o)),
                        curExpr.lastIndexOf('('));
    curExpr = li >= 0 ? curExpr.slice(0, li+1) + curDisp : curDisp;
  }
  renderCalc();
}

function renderCalc() {
  // 히스토리
  const hEl = document.getElementById('calcHistory');
  const sep  = document.getElementById('historySep');
  if (calcHist.length > 0) {
    hEl.innerHTML = calcHist.map((h,i) =>
      `<div class="history-line" onclick="useHist(${i})">${h.expr} = ${h.result}</div>`
    ).join('');
    sep.style.display = 'block';
  } else { hEl.innerHTML = ''; sep.style.display = 'none'; }

  // 수식 표시
  document.getElementById('calcExpr').innerHTML =
    colorExpr(evaled ? curDisp : (curExpr || '0'));

  // 실시간 미리보기
  let preview = '';
  if (!evaled && curExpr && !OPS.includes(curExpr.slice(-1)) && curExpr !== curDisp) {
    const r = tryEval(curExpr);
    if (r !== null) preview = '= ' + numFmt(r);
  }
  document.getElementById('calcPreview').textContent = preview;
}

function useHist(i) {
  const h = calcHist[i];
  lastRes = parseFloat(h.result.replace(/,/g,''));
  curDisp = h.result; curExpr = ''; evaled = true;
  renderCalc();
}

renderCalc();

function formatWithCommas(el, allowDecimal) {
  let v = el.value.replace(/,/g, '');
  if (allowDecimal) {
    v = v.replace(/[^\d.]/g, '');
    const firstDot = v.indexOf('.');
    if (firstDot !== -1) {
      v = v.slice(0, firstDot + 1) + v.slice(firstDot + 1).replace(/\./g, '');
    }
    const [intPart, decPart] = v.split('.');
    const fmtInt = (intPart || '').replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    el.value = decPart !== undefined ? fmtInt + '.' + decPart : fmtInt;
  } else {
    v = v.replace(/\D/g, '');
    el.value = v.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
  }
}
function numVal(el) {
  return parseFloat(el.value.replace(/,/g, '')) || 0;
}

let fxRates = {};

async function loadFX() {
  try {
    const res = await fetch('https://open.er-api.com/v6/latest/KRW');
    const d   = await res.json();
    fxRates = d.rates;
    const dt = new Date(d.time_last_update_utc);
    document.getElementById('fxStatusText').textContent =
      `${dt.getMonth()+1}/${dt.getDate()} ${dt.getHours()}시 기준 환율`;
    document.querySelector('#screen-fx .rate-status i').className = 'ti ti-check';
    convertFX();
    renderQuick();
  } catch(e) {
    document.getElementById('fxStatusText').textContent = '환율 로드 실패 (오프라인 모드)';
  }
}

function convertFX() {
  if (!Object.keys(fxRates).length) return;
  const from = document.getElementById('fromCur').value;
  const to   = document.getElementById('toCur').value;
  const val  = numVal(document.getElementById('fromVal'));
  const rate = (1 / fxRates[from]) * fxRates[to];
  const result = val * rate;
  document.getElementById('toVal').value =
    result !== 0 && Math.abs(result) < 0.01
      ? result.toFixed(6)
      : numFmt(parseFloat(result.toFixed(2)));
  const r1 = rate < 0.01 ? rate.toFixed(6) : rate.toFixed(4);
  document.getElementById('rateInfo').textContent = `1 ${from} = ${r1} ${to}`;
}

function swapFX() {
  const f = document.getElementById('fromCur');
  const t = document.getElementById('toCur');
  [f.value, t.value] = [t.value, f.value];
  convertFX();
}

function loadUnitSelects() {
  const def  = UNITS[curUnitType];

  document.getElementById('fromUnit').innerHTML = opts;

function convertUnit() {
  const from = document.getElementById('fromUnit').value;
  const to   = document.getElementById('toUnit').value;
  const val  = parseFloat(document.getElementById('fromUnitVal').value) || 0;
  const def  = UNITS[curUnitType];
  let result;
  if (curUnitType === 'temp') {
    let c = from==='C' ? val : from==='F' ? (val-32)*5/9 : val-273.15;
    result = to==='C' ? c : to==='F' ? c*9/5+32 : c+273.15;
  } else {
    const fu = def.units.find(u => u.k === from);
    const tu = def.units.find(u => u.k === to);
    result = val * fu.base / tu.base;
  }
  const disp = Math.abs(result) >= 0.0001 || result === 0
    ? parseFloat(result.toPrecision(8)).toString()
    : result.toExponential(4);
  document.getElementById('toUnitVal').value = disp;
  if (curUnitType !== 'temp') {
    const fu = UNITS[curUnitType].units.find(u => u.k === from);
    const tu = UNITS[curUnitType].units.find(u => u.k === to);
    const r  = parseFloat((fu.base / tu.base).toPrecision(6));
    document.getElementById('unitEqText').textContent = `1 ${from} = ${r} ${to}`;
  } else {
    document.getElementById('unitEqText').textContent = '';
  }
}

function calcAge() {
  const y = parseInt(document.getElementById('birthYear').value, 10);
  const m = parseInt(document.getElementById('birthMonth').value, 10);
  const d = parseInt(document.getElementById('birthDay').value, 10);
  if (!y || !m || !d || y < 1000 || m < 1 || m > 12 || d < 1 || d > 31) return;
  const b = new Date(y, m - 1, d);
  if (b.getFullYear() !== y || b.getMonth() !== m - 1 || b.getDate() !== d) return;
  const t = new Date();
  const by = b.getFullYear(), bm = b.getMonth(), bd = b.getDate();
  const ty = t.getFullYear(), tm = t.getMonth(), td = t.getDate();
  const intAge = ty - by - (tm < bm || (tm === bm && td < bd) ? 1 : 0);
  const korAge = ty - by + 1;
  const diffDays = Math.floor((t - b) / 86400000);
  const nextBday = new Date(ty, bm, bd);
  if (nextBday <= t) nextBday.setFullYear(ty + 1);
  const daysLeft = Math.ceil((nextBday - t) / 86400000);
  document.getElementById('korAge').textContent    = korAge + '세';
  document.getElementById('intAge').textContent    = intAge + '세';
  document.getElementById('ageBirth').textContent  = `${by}.${String(bm+1).padStart(2,'0')}.${String(bd).padStart(2,'0')}`;
  document.getElementById('ageToday').textContent  = `${ty}.${String(tm+1).padStart(2,'0')}.${String(td).padStart(2,'0')}`;
  document.getElementById('ageDays').textContent   = diffDays.toLocaleString() + '일';
  document.getElementById('ageNextBday').textContent = daysLeft + '일';
  document.getElementById('ageResults').classList.remove('hidden');
}

function calcOA() {
  const cost  = numVal(document.getElementById('oa_cost'));
  const daily = numVal(document.getElementById('oa_daily'));
  const slots = numVal(document.getElementById('oa_slots'));
  const cpe         = daily > 0 ? cost / (daily * 30) : 0;
  const monthlyRev  = cost * slots;
  const yearlyRev   = monthlyRev * 12;
  const fmt = v => v > 0 ? v.toLocaleString('ko-KR', {maximumFractionDigits:1}) : '–';
  document.getElementById('oa_cpe').textContent     = daily > 0 ? fmt(cpe) : '–';
  document.getElementById('oa_monthly').textContent = slots > 0 && cost > 0 ? monthlyRev.toLocaleString('ko-KR') : '–';
  document.getElementById('oa_yearly').textContent  = slots > 0 && cost > 0 ? yearlyRev.toLocaleString('ko-KR')  : '–';
}
